# Сериализация и десериализация

Часто используется как синоним маршалинга/демаршалинга, но с акцентом на сохранение _состояния_ объекта, включая его тип и структуру.

```js
class User {
    constructor(name, createdAt) {
        this.name = name;
        this.createdAt = createdAt;
    }
}

const user = new User("Alice", new Date("2024-01-01"));

// СЕРИАЛИЗАЦИЯ — сохраняем состояние
const serialized = JSON.stringify({
    __class: "User",
    name: user.name,
    createdAt: user.createdAt.toISOString()
});
// "{"__class":"User","name":"Alice","createdAt":"2024-01-01T00:00:00.000Z"}"

// ДЕСЕРИАЛИЗАЦИЯ — восстанавливаем объект с правильным классом
function deserializeUser(data) {
    const obj = JSON.parse(data);
    if (obj.__class === "User") {
        return new User(obj.name, new Date(obj.createdAt));
    }
    throw new Error("Unknown type");
}

const restored = deserializeUser(serialized);
console.log(restored instanceof User);  // true
console.log(restored.createdAt.getMonth());  // 0 (работает как Date)
```
